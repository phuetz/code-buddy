Le contrôle de référence est basé sur `git grep` et vérifie qu’aucun lien Markdown ne cible encore un nom de fichier déplacé à la racine ; les noms dans les titres et les fixtures synthétiques ne sont pas des chemins cassés.

Script de preuve exécuté (aucun fichier temporaire hors du clone) :

```sh
set -u
root_output=$(git ls-files '*.md' | grep -v /)
root_report_count=$(printf '%s\n' "$root_output" | rg -c '^(AUDIT-|BANC-|BILAN-|RAPPORT-|REPARATION-|REVUE-)' || true)
test "${root_report_count:-0}" -eq 0
broken=0
while IFS=: read -r file line rest; do
  while IFS= read -r target; do
    target=${target#*](}; target=${target%)}; target=${target%%[[:space:]]*}
    case "$target" in ''|'#'*|'http://'*|'https://'*|'mailto:'*|'tel:'*) continue;; esac
    clean=${target%%#*}; clean=${clean%%\?*}; [ -n "$clean" ] || continue
    resolved=$(realpath -m --relative-to=. "$(dirname "$file")/$clean")
    [ -e "$resolved" ] || broken=$((broken + 1))
  done < <(printf '%s\n' "$rest" | rg -o '\]\([^)]*\)' || true)
done < <(git grep -n -E '\]\([^)]*\)' -- 'docs/reports/**/*.md' 'docs/briefs/**/*.md' 'docs/research/**/*.md' 'docs/operations/**/*.md' 'docs/RELEASING.md' 2>/dev/null || true)
test "$broken" -eq 0
printf 'RANG1 report refs: root reports = %s, broken moved-doc links = %s\n' "${root_report_count:-0}" "$broken"
```
